    def clean(self):
        d = super().clean()
        fecha = d.get('fecha')
        hora = d.get('hora')
        medico = d.get('medico')
        especialidad = d.get('especialidad')
        paciente = d.get('paciente')
        motivo = d.get('motivo', '')

        if len(motivo.strip()) < 20:
            self.add_error('motivo',
                'El motivo debe tener al menos 20 caracteres.')
        if fecha and fecha < timezone.localdate():
            self.add_error('fecha', 'No se permiten fechas pasadas.')
        if medico and especialidad and medico.especialidad_id != especialidad.id:
            self.add_error('medico',
                'El médico no pertenece a esta especialidad.')

        if fecha and hora:
            weekday = fecha.weekday()
            apertura, cierre = (time(8), time(13)) if weekday == 5 else (time(8), time(18))
            if weekday == 6 or not apertura <= hora < cierre:
                self.add_error('hora', 'Fuera del horario institucional.')
            if medico and not medico.horarios.filter(
                dia_semana=weekday,
                hora_inicio__lte=hora,
                hora_fin__gt=hora
            ).exists():
                self.add_error('hora',
                    'Fuera del horario individual del médico.')
        return d
